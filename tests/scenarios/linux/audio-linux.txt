# Linux 6.18's ALSA through LinuxKPI (kernel built with --features
# linux-sound): snd-hda-intel with the generic codec on QEMU's intel-hda,
# with a recording on a second controller, and snd-usb-audio on QEMU's usb-audio (isochronous OUT URBs). RustOS's
# audio tools use ALSA's OSS emulation (/dev/dsp, /dev/dsp1); the host
# checks the tones QEMU recorded.
wait # $ 240
send ls /dev/snd | tr '\n' ' '; echo
wait pcmC0D0p
wait pcmC1D0c
wait pcmC2D0p
wait # $
send cat /proc/asound/cards
wait \[Intel\s*\]: HDA-Intel
wait USB-Audio - QEMU USB Audio
wait # $
send mixer -d /dev/mixer; mixer -d /dev/mixer volume 90
wait volume\s+90% 90%
wait # $
send beep -f 1000 -l 1500 -v 60 -d /dev/dsp; echo hda-beeped
wait hda-beeped\r 60
wait # $
send beep -f 2000 -l 1500 -v 60 -d /dev/dsp2; echo usb-beeped
wait usb-beeped\r 60
wait # $
# Recording from the second card's microphone: a second of 48 kHz
# stereo 16-bit.
send rec -d /dev/dsp1 -t 1 /tmp/r.wav && ls -l /tmp/r.wav
wait 192000 bytes recorded 30
wait # $
monitor quit
sleep 2
hostrun python3 "$(git rev-parse --show-toplevel)/tools/check-wav.py" /tmp/rustos-audio-lhda.wav 1000
hostrun python3 "$(git rev-parse --show-toplevel)/tools/check-wav.py" /tmp/rustos-audio-lusb.wav 2000
