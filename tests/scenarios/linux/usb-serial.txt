# Linux usb-serial + ftdi_sio through LinuxKPI on QEMU's usb-serial (an
# FTDI FT232; kernel built with `--features linux-serial`): /dev/ttyUSB0 is
# a RustOS terminal. tools/serial-echo.py on the far end of the line sends
# back what it gets, upper-cased. QEMU plugs the adapter in when that
# helper connects to the line.
wait # $ 240
hostbg rm -f /tmp/rustos-usbser.log; exec python3 tools/serial-echo.py /tmp/rustos-usbser.sock /tmp/rustos-usbser.log
wait /dev/ttyUSB0 60
send true
wait # $
send dmesg | grep -i "ftdi\|usbserial\|ttyUSB" | cat
wait FTDI USB Serial Device converter now attached to ttyUSB0
wait # $
send ls -l /dev/ttyUSB0
wait 188, *0
wait # $
# Raw by default: what comes back is exactly what the far end sent.
send cat /dev/ttyUSB0 > /tmp/rx & sleep 1; echo hello-serial > /dev/ttyUSB0; sleep 2; cat /tmp/rx
wait HELLO-SERIAL
wait # $
hostrun grep -q hello-serial /tmp/rustos-usbser.log
# Line settings reach the driver (stty sets 115200 baud through TCSETS).
send stty -F /dev/ttyUSB0 115200 && stty -F /dev/ttyUSB0 | head -1
wait speed 115200 baud
wait # $
send echo second-line > /dev/ttyUSB0; sleep 2; cat /tmp/rx
wait SECOND-LINE
wait # $
# Unplugged while open: the reader sees end of file and the node goes away.
monitor device_del usbser
wait usb1-1: disconnected 30
send sleep 1; jobs; ls /dev | grep -c ttyUSB | sed s/^/count=/
wait count=0
wait # $
