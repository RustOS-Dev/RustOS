# LinuxKPI's character-device bridge: /dev/kpi-test (src/linuxkpi/c/testdev.c,
# kernel built with --features linux-test) driven from user space by
# musl-kpitest: read/write, ioctl with user pointers, poll wake-ups, mmap
# through remap_pfn_range and through a fault handler.
wait # $ 240
send ls -l /dev/kpi-test
wait 10, \d+
wait # $
send musl-kpitest
wait kpitest: \d+ passed, 0 failed 60
wait # $
send ls /sys/class/misc/kpi-test; cat /sys/class/misc/kpi-test/dev
wait 10:\d+
wait # $
