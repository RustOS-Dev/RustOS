# OpenGL ES through Mesa on KMS (M41): kmscube draws a spinning cube with
# EGL/GBM and GLES2 on /dev/dri/card0 (QEMU standard VGA, Linux bochs),
# rendered by llvmpipe (then softpipe) through kms_swrast, which needs no
# GPU. Needs the weston port in the image (tools/install-port.sh
# --initramfs weston) and
# a kernel built with --features linux-drivers. The screendump shows
# kmscube's grey clear colour with the cube on it (neither the console nor
# an empty clear).
hostrun rm -rf /tmp/rustos-gl && mkdir -p /tmp/rustos-gl
wait # $ 240
# kmscube writes to the serial console (a tty, so line by line): it
# reports its frame rate every 2 s once frames flow. llvmpipe compiles its
# shaders before the first frame, which takes a while under TCG.
send kmscube -D /dev/dri/card0 -c 100000 &
wait renderer: "llvmpipe 120
wait Rendered \d+ frames 150
monitor screendump /tmp/rustos-gl/cube.ppm
hostrun python3 "$(git rev-parse --show-toplevel)/tools/check-png.py" /tmp/rustos-gl/cube.ppm 'count:128,128,128>=500000' 'count:128,128,128<=1000000'
send kill $!; wait; echo kms''cube-done
wait kmscube-done
wait # $
# softpipe, the interpreting renderer (GALLIUM_DRIVER=softpipe).
send GALLIUM_DRIVER=softpipe kmscube -D /dev/dri/card0 -c 20 > /tmp/kmscube2.txt 2>&1; cat /tmp/kmscube2.txt
wait renderer: "softpipe" 180
wait Rendered \d+ frames 60
wait # $
# The Vulkan loader finds Mesa's RADV and ANV drivers (there is no AMD or
# Intel GPU here, so neither has a device to offer).
send VK_LOADER_DEBUG=driver vulkaninfo --summary 2>&1 | grep -a -c "Found ICD manifest file .*/\(radeon\|intel\)_icd" | sed 's/^/icds=/'
wait icds=[1-9] 120
wait # $
send vulkaninfo --summary 2>&1 | grep -a -e "Failed to detect any valid GPUs" -e "GPU0"
wait Failed to detect any valid GPUs 60
wait # $
