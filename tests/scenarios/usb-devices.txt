# HID report protocol (QEMU usb-tablet) and USB Attached SCSI.
wait # $ 240
# (The two UAS devices enumerate in either order: one grep each.)
send sleep 2; dmesg | grep "SCSI, .* streams" | cat
wait USB Attached SCSI, [0-9]+ streams, [0-9]+ commands queued
wait # $
send dmesg | grep "SCSI (no streams" | cat
wait USB Attached SCSI \(no streams
wait # $
send dmesg | grep "HID tablet" | cat
wait HID tablet \(report protocol\)
wait # $
# Tablet buttons and absolute axes reach /dev/input/event0.
send evtest -c 4 > /tmp/ev.txt &
wait # $
monitor mouse_set 3
monitor mouse_button 1
monitor mouse_button 0
sleep 2
send cat /tmp/ev.txt
wait event KEY code 0x110 value 1\r\n
wait event ABS code 0x0
wait event KEY code 0x110 value 0\r\n
wait # $
# The UAS disk (ext4) is automounted; write, verify, remount.
send d=$(grep " ext4 " /proc/mounts | cut -d" " -f2); echo disk=$d; dd if=/dev/urandom of=$d/r.bin bs=65536 count=64 2>/dev/null; sha256sum $d/r.bin > /tmp/r.sum; seq 1 20000 > $d/n.txt; sync; wc -l $d/n.txt
wait disk=/mnt/sd
wait 20000
wait # $
send dev=$(grep " ext4 " /proc/mounts | cut -d" " -f1); umount $d && mount $dev /mnt/uas && sha256sum /mnt/uas/r.bin | cut -c1-64 > /tmp/b; cut -c1-64 /tmp/r.sum > /tmp/a; cmp /tmp/a /tmp/b && echo hash-same; tail -1 /mnt/uas/n.txt
wait hash-same
wait 20000\r\n
wait # $
send umount /mnt/uas && echo unmounted
wait unmounted
wait # $
# The USB 2 UAS disk (ext2): the same through READ/WRITE READY.
send e=$(grep " ext2 " /proc/mounts | cut -d" " -f2); echo disk2=$e; dd if=/dev/urandom of=$e/r2.bin bs=65536 count=32 2>/dev/null; sha256sum $e/r2.bin | cut -c1-64 > /tmp/a2; sync; dev2=$(grep " ext2 " /proc/mounts | cut -d" " -f1); umount $e && mount $dev2 /mnt/u2 && sha256sum /mnt/u2/r2.bin | cut -c1-64 | cmp - /tmp/a2 && echo hs-same; umount /mnt/u2
wait disk2=/mnt/sd
wait hs-same
wait # $
hostrun e2fsck -fn "$RUSTOS_DISK0"
hostrun e2fsck -fn "$RUSTOS_DISK1"
