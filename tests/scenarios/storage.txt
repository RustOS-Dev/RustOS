# FAT: long names, nested dirs, append, rename, delete
send mkdir -p "/storage/Long Directory Name/sub" && echo first > "/storage/Long Directory Name/sub/a file.txt"
wait # $
send echo second >> "/storage/Long Directory Name/sub/a file.txt"; cat "/storage/Long Directory Name/sub/a file.txt"
wait first\r\nsecond
wait # $
send seq 1 20000 > /storage/big.txt; wc -l /storage/big.txt; sha256sum /storage/big.txt

wait # $
send mv /storage/big.txt "/storage/Long Directory Name/Big Renamed.TXT" && ls "/storage/Long Directory Name"
wait Big Renamed.TXT
wait # $
send rm "/storage/Long Directory Name/sub/a file.txt" && rmdir "/storage/Long Directory Name/sub" && ls "/storage/Long Directory Name"

wait # $
# ext2 on NVMe (mounted at /mnt/nvme0n1)

wait # $
send mkdir /mnt/nvme0n1/d && seq 1 5000 > /mnt/nvme0n1/d/n && ln -s d/n /mnt/nvme0n1/link && cat /mnt/nvme0n1/link | tail -1
wait 5000\r\n
wait # $
send ln /mnt/nvme0n1/d/n /mnt/nvme0n1/hard; stat /mnt/nvme0n1/hard | grep -i links
wait # $
send mv /mnt/nvme0n1/hard /mnt/nvme0n1/d/moved && rm /mnt/nvme0n1/d/n && wc -l /mnt/nvme0n1/d/moved
wait 5000 /mnt/nvme0n1/d/moved
wait # $
# ext4 on virtio: read-only
